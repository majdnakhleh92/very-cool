local function toboolan(str)
	local String = tostring(str):lower()

	if String == "true" then
		return true
	else
		return false
	end
end

local function ParseNumber(str)
	str = tostring(str):gsub("%s+", ""):lower()

	local number, suffix = string.match(str, "^(%d*%.?%d+)([kmb]?)$")

	if not number then
		return nil
	end

	local value = tonumber(number)
	if not value then
		return nil
	end

	if suffix == "k" then
		value = value * 1000
	elseif suffix == "m" then
		value = value * 1000000
	elseif suffix == "b" then
		value = value * 1000000000
	end

	return value
end

local function Color3ToHex(color)
	local r = math.floor(color.R * 255)
	local g = math.floor(color.G * 255)
	local b = math.floor(color.B * 255)
	return bit32.bor(bit32.lshift(r, 16), bit32.lshift(g, 8), b)
end

local function HexToColor3(hex)
	local r = bit32.rshift(bit32.band(hex, 0xFF0000), 16) / 255
	local g = bit32.rshift(bit32.band(hex, 0x00FF00), 8) / 255
	local b = bit32.band(hex, 0x0000FF) / 255
	return Color3.new(r, g, b)
end

local function CreateMutationPage(i)
	if i == 0 or i > 12 then return end
	if not Settings["Mutations"] then
		Settings["Mutations"] = {}
		for j = 1, 12 do
			table.insert(Settings["Mutations"], {})
		end
	end

	local index = Settings["Mutations"][i]

	configurations:Label("Mutation Page " .. i)

	local targetMutation = ""

	local function getMatchingBrainrots()
		local result = {}
		if targetMutation == "" then return result end
		if not Settings["SpecificatedPets"] then return result end
		for petName, petData in Settings["SpecificatedPets"] do
			if type(petData) ~= "table" then continue end
			if petData.MutWhitelist and table.find(petData.MutWhitelist, targetMutation) then
				Settings["SpecificatedPets"][petName].Enabled = true
				table.insert(result, petName)
			end
		end
		return result
	end

	local targetedBrainrots = configurations:Paragraph({
		Title = "Brainrots with this mutation (Page " .. i .. ")",
		Body = "",
	})

	local mutationTarget = configurations:TextBox({
		Name = "Target Mutation",
		Callback = function(value)
			index["MutationTarget"] = value
			targetMutation = value
			updateParagraph(targetedBrainrots, getMatchingBrainrots())
		end,
		OnlyNumbers = false,
		OnLeave = true,
		ClearTextOnFocus = false,
		PlaceHolderText = "Cursed, Radioactive, etc"
	})

	local targetBrainrotTextbox = configurations:TextBox({
		Name = "Target Brainrot",
		Callback = function(value)
			if targetMutation == "" then Notification("No Target Mutation!", "Please enter a target mutation first") return end
			if not Settings["SpecificatedPets"] then Settings["SpecificatedPets"] = {} end

			local trimmed = value:match("^%s*(.-)%s*$")
			if trimmed == "" then return end

			for entry in trimmed:gmatch("[^,]+") do
				local cleaned = entry:match("^%s*(.-)%s*$")
				if cleaned ~= "" then
					if not Settings["SpecificatedPets"][cleaned] then
						Settings["SpecificatedPets"][cleaned] = {}
					end
					local pet = Settings["SpecificatedPets"][cleaned]
					pet.MutWhitelist = pet.MutWhitelist or {}

					if not table.find(pet.MutWhitelist, targetMutation) then
						table.insert(pet.MutWhitelist, targetMutation)
						pet.Enabled = true
						Notification("Brainrot included!", "[ " .. cleaned .. " ]")
					else
						Notification("Already included!", "[ " .. cleaned .. " ]")
					end
				end
			end

			updateParagraph(targetedBrainrots, getMatchingBrainrots())
		end,
		OnLeave = true,
		PlaceHolderText = "Brainrot1 or brainrot1, brainrot2, etc"
	})

	local detargetTextbox = configurations:TextBox({
		Name = "Detarget Brainrots",
		Callback = function(value)
			if targetMutation == "" then Notification("No Target Mutation!", "Please enter a target mutation first") return end
			if not Settings["SpecificatedPets"] then return end

			local trimmed = value:match("^%s*(.-)%s*$")
			if trimmed == "" then return end

			for entry in trimmed:gmatch("[^,]+") do
				local cleaned = entry:match("^%s*(.-)%s*$")
				if cleaned ~= "" then
					local pet = Settings["SpecificatedPets"][cleaned]
					if pet and pet.MutWhitelist then
						local found = table.find(pet.MutWhitelist, targetMutation)
						if found then
							table.remove(pet.MutWhitelist, found)
							Notification("Removed!", "[ " .. cleaned .. " ]")
						else
							Notification("Not in targeted brainrots!", "[ " .. cleaned .. " ]")
						end
					else
						Notification("Not in targeted brainrots!", "[ " .. cleaned .. " ]")
					end
				end
			end

			updateParagraph(targetedBrainrots, getMatchingBrainrots())
		end,
		OnLeave = true,
		PlaceHolderText = "Brainrot1 or brainrot1, brainrot2, etc"
	})

	if index["MutationTarget"] then
		mutationTarget:SetInput(index["MutationTarget"])
		targetMutation = index["MutationTarget"]
		updateParagraph(targetedBrainrots, getMatchingBrainrots())
	end

	configurations:Divider()
end
